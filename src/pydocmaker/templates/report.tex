% pydocmaker report template (LaTeX)
%
% the LaTeX counterpart of "report.typ": a more comprehensive report template
% with a title page, table of contents, list of figures / tables, acronyms,
% applicable and reference documents and configurable headers / footers
% (front matter in roman page numbers, body in arabic page numbers).
%
% copied and modified from https://github.com/roland-KA/basic-report-typst-template/tree/main
% (via report.typ) THANKS!
%
% NOTE: pydocmaker parameter names double as LaTeX macro names here, so they
% must be letters only (no digits). The logo parameter of report.typ
% ("pLogoB64", a base64 string) is therefore "pLogoFile" in this template:
% pass an image FILE NAME (e.g. "logo.png") and provide the file as an
% attachment. pydocmaker rewrites file-path parameter values to their bare
% basename and attaches the file automatically, e.g.:
%
%     pydocmaker build my_report.md -t report.tex --pLogoFile my_logo.png
%
% When no logo file is given, the pydocmaker logo is shown instead (unless
% pHidePydocmaker is set), provided a "pydocmaker_logo.png" attachment is
% available for the build (e.g. via "report.assets/pydocmaker_logo.png" in a
% registered template directory or "-a pydocmaker_logo.png"). Missing logo
% files are skipped gracefully.

% ===== helper macros ====================

% \pdmifempty{\macro}{<if empty/undefined>}{<if not empty>}
\newcommand{\pdmifempty}[3]{%
  \if\relax\detokenize\expandafter{#1}\relax
    \expandafter\@firstoftwo
  \else
    \expandafter\@secondoftwo
  \fi
  {#2}{#3}%
}

% \pdmifflag{\macro}{<if true>}{<if false>}
% a flag parameter is "on" when its macro content is "true" or "1"
\newcommand{\pdmifflag}[3]{%
  \ifdefined#1%
    \edef\pdm@flagval{\detokenize\expandafter{#1}}%
  \else
    \edef\pdm@flagval{}%
  \fi
  \edef\pdm@onetest{\detokenize{1}}%
  \ifx\pdm@flagval\pdm@onetest
    \edef\pdm@flagval{\detokenize{true}}%
  \fi
  \edef\pdm@truetest{\detokenize{true}}%
  \ifx\pdm@flagval\pdm@truetest
    \expandafter\@firstoftwo
  \else
    \expandafter\@secondoftwo
  \fi
  {#2}{#3}%
}

% render a pydocmaker key-value macro (a description environment whose items
% look like "\item{key}{value}") with bold labels.
% \pdmkvbegin{prefix} <macro> \pdmkvend
% - empty prefix: the keys are the (bold) labels (used for the acronyms)
% - prefix given (e.g. "AD" or "RD"): items are numbered prefix1, prefix2, ...
\newcounter{pdm@kv}
\newcommand{\pdmkvbegin}[1]{%
  \begingroup
    \def\pdm@prefix{#1}%
    \setcounter{pdm@kv}{0}%
    \let\pdm@origitem\item
    \def\item##1{%
      \if\relax\detokenize\expandafter{\pdm@prefix}\relax
        \pdm@origitem[\textbf{##1}]%
      \else
        \stepcounter{pdm@kv}%
        \pdm@origitem[\textbf{\pdm@prefix\arabic{pdm@kv}}]%
      \fi
      \ignorespaces
    }%
}
\newcommand{\pdmkvend}{\par\endgroup}

% the pydocmaker logo (looked up by name; the macro indirection keeps
% template file-reference scanning from picking the name up)
\newcommand{\pdmpydlogo}{pydocmaker_logo.png}

% the fixed rows of the title page info block (Author / Date)
\newcommand{\pdm@inforows}{%
  \pdmifempty{\pAuthor}{}{Author: & \pAuthor \\}%
  \pdmifempty{\pDate}{%
    \pdmifflag{\pHideDate}{}{Date: & \today \\}%
  }{%
    Date: & \pDate \\%
  }%
}

% the title page info block: the fixed rows plus the pTitlepageInfoDict
% entries, rendered as one aligned two column table
\newcommand{\pdmtitleinfo}{%
  \begingroup
    \renewenvironment{description}{%
      \begin{tabular}{r@{\hspace{2.5em}}l}%
      \pdm@inforows
    }{%
      \end{tabular}%
    }%
    \renewcommand{\item}[2]{##1 & ##2 \\}%
    \pdmifempty{\pTitlepageInfoDict}{%
      \begin{tabular}{r@{\hspace{2.5em}}l}\pdm@inforows\end{tabular}%
    }{%
      \pTitlepageInfoDict
    }%
  \endgroup
}

% localize the table of contents title (report.typ translates the outline
% title per language; scripts that pdflatex cannot render fall back to English)
\newcommand{\pdmapplylang}{%
  \edef\pdm@lang{\detokenize\expandafter{\pLanguage}}%
  \edef\pdm@x{\detokenize{de}}\ifx\pdm@lang\pdm@x \renewcommand{\contentsname}{Inhalt}\fi
  \edef\pdm@x{\detokenize{fr}}\ifx\pdm@lang\pdm@x \renewcommand{\contentsname}{Table des mati\`eres}\fi
  \edef\pdm@x{\detokenize{es}}\ifx\pdm@lang\pdm@x \renewcommand{\contentsname}{Contenido}\fi
  \edef\pdm@x{\detokenize{it}}\ifx\pdm@lang\pdm@x \renewcommand{\contentsname}{Indice}\fi
  \edef\pdm@x{\detokenize{nl}}\ifx\pdm@lang\pdm@x \renewcommand{\contentsname}{Inhoud}\fi
  \edef\pdm@x{\detokenize{pt}}\ifx\pdm@lang\pdm@x \renewcommand{\contentsname}{\'Indice}\fi
}
